import React, { useEffect, useState } from 'react';
import { ActivityIndicator, StyleSheet, Text, TextInput, TouchableOpacity, View } from 'react-native';
import { NativeStackScreenProps } from '@react-navigation/native-stack';
import { Ionicons } from '@expo/vector-icons';
import ScreenContainer from '../../components/common/ScreenContainer';
import Button from '../../components/common/Button';
import RewardsModal from '../../components/trip/RewardsModal';
import { useTrip } from '../../contexts/TripContext';
import { awardPointsForTrip } from '../../services/pointsStore';
import { setFavoriteDriver } from '../../services/favoriteDriverStore';
import { ratingsRemote } from '../../services/extrasApi';
import { syncQuiet } from '../../services/remoteSync';
import { tripsApi } from '../../services/api';
import { PaymentState } from '../../types';
import { HomeStackParamList } from '../../navigation/types';
import { colors, radius, spacing, typography } from '../../constants/theme';

type Props = NativeStackScreenProps<HomeStackParamList, 'TripCompleted'>;

const PAYMENT_LABELS: Record<PaymentState, string> = {
  processing: 'A processar pagamento...',
  confirmed: 'Pagamento confirmado',
  failed: 'Falha no pagamento - tenta outro metodo',
};

export default function TripCompletedScreen({ navigation }: Props) {
  const { activeTrip, clearActiveTrip } = useTrip();
  const [rating, setRating] = useState(5);
  const [comment, setComment] = useState('');
  const [pointsEarned, setPointsEarned] = useState<number | null>(null);
  const [rewardsVisible, setRewardsVisible] = useState(false);
  const [favoriteSaved, setFavoriteSaved] = useState(false);
  // Estado real do pagamento (seccao 29 do documento): nunca assumimos
  // confirmacao instantanea - simulamos "a processar" antes de confirmar.
  const [paymentState, setPaymentState] = useState<PaymentState>('processing');

  useEffect(() => {
    if (!activeTrip) return;
    let alive = true;
    awardPointsForTrip(activeTrip.id, activeTrip.priceKz).then((points) => {
      if (alive) setPointsEarned(points);
    });
    // Sem ligacao ao servidor (viagem so local): mantem a simulacao.
    if (!activeTrip.remoteId) {
      const timer = setTimeout(() => {
        if (alive) setPaymentState('confirmed');
      }, 1800);
      return () => {
        alive = false;
        clearTimeout(timer);
      };
    }
    // Com servidor: pagamento por carteira e debitado automaticamente (o saldo
    // ja desconta a viagem concluida). Se nao houver saldo, o servidor devolve
    // paymentConfirmed=false e mostramos a falha em vez de inventar sucesso.
    tripsApi
      .get(activeTrip.remoteId)
      .then((t: any) => {
        if (alive) setPaymentState(t.paymentConfirmed === false ? 'failed' : 'confirmed');
      })
      .catch(() => {
        if (alive) setPaymentState('failed');
      });
    return () => {
      alive = false;
    };
  }, [activeTrip]);

  useEffect(() => {
    if (!activeTrip || !activeTrip.driver) {
      navigation.popToTop();
    }
  }, [activeTrip, navigation]);

  if (!activeTrip || !activeTrip.driver) {
    return null;
  }

  // Selo de Trajeto Limpo: compara a distancia estimada com a distancia real
  // percorrida (rota OSRM). Diferencas grandes podem indicar desvio de rota.
  const hasCleanRoute =
    activeTrip.actualDistanceKm !== undefined &&
    Math.abs(activeTrip.actualDistanceKm - activeTrip.distanceKm) <= activeTrip.distanceKm * 0.15;

  const handleFinish = () => {
    if (activeTrip.driver) {
      syncQuiet(() => ratingsRemote.rate({
        tripRef: activeTrip.id,
        driverId: activeTrip.driver!.id,
        stars: rating,
        comment: comment.trim() || undefined,
      }));
    }
    clearActiveTrip();
    navigation.popToTop();
  };

  const handleMarkFavorite = async () => {
    if (!activeTrip.driver) return;
    await setFavoriteDriver(activeTrip.driver.id, activeTrip.driver.fullName);
    setFavoriteSaved(true);
  };

  return (
    <ScreenContainer style={styles.container}>
      <View style={styles.checkBadge}>
        <Ionicons name="checkmark" size={36} color={colors.textInverse} />
      </View>
      <Text style={styles.title}>Viagem concluida!</Text>
      <Text style={styles.subtitle}>Obrigado por viajar com o AngoTour.</Text>

      <View style={styles.priceCard}>
        <Text style={styles.priceLabel}>Total da viagem</Text>
        <Text style={styles.priceValue}>{activeTrip.priceKz.toLocaleString('pt-AO')} Kz</Text>
        <View style={styles.protectedRow}>
          <Ionicons name="shield-checkmark" size={14} color={colors.primary} />
          <Text style={styles.protectedText}>Preco protegido - sem alteracoes desde a confirmacao</Text>
        </View>

        <View style={styles.paymentRow}>
          {paymentState === 'processing' ? (
            <ActivityIndicator size="small" color={colors.textSecondary} />
          ) : (
            <Ionicons
              name={paymentState === 'confirmed' ? 'checkmark-circle' : 'alert-circle'}
              size={14}
              color={paymentState === 'confirmed' ? colors.primary : colors.danger}
            />
          )}
          <Text style={[styles.paymentText, paymentState === 'failed' && { color: colors.danger }]}>
            {PAYMENT_LABELS[paymentState]}
          </Text>
        </View>

        {hasCleanRoute ? (
          <View style={styles.cleanRouteRow}>
            <Ionicons name="ribbon-outline" size={14} color={colors.teal} />
            <Text style={styles.cleanRouteText}>Trajeto Limpo - percurso seguiu a rota esperada</Text>
          </View>
        ) : null}
      </View>

      {pointsEarned !== null && pointsEarned > 0 ? (
        <TouchableOpacity style={styles.pointsCard} activeOpacity={0.85} onPress={() => setRewardsVisible(true)}>
          <Ionicons name="star" size={20} color={colors.primaryDark} />
          <View style={{ flex: 1, marginLeft: spacing.sm }}>
            <Text style={styles.pointsTitle}>+{pointsEarned} AngoPoints ganhos</Text>
            <Text style={styles.pointsSubtitle}>Toca para ver o catalogo de recompensas</Text>
          </View>
          <Ionicons name="chevron-forward" size={18} color={colors.primaryDark} />
        </TouchableOpacity>
      ) : null}

      <Text style={styles.sectionTitle}>Avalie a sua viagem</Text>
      <View style={styles.starsRow}>
        {[1, 2, 3, 4, 5].map((star) => (
          <TouchableOpacity key={star} onPress={() => setRating(star)}>
            <Ionicons
              name={star <= rating ? 'star' : 'star-outline'}
              size={32}
              color={colors.amber}
              style={{ marginHorizontal: 4 }}
            />
          </TouchableOpacity>
        ))}
      </View>

      {rating === 5 && !favoriteSaved ? (
        <TouchableOpacity style={styles.favoriteButton} activeOpacity={0.85} onPress={handleMarkFavorite}>
          <Ionicons name="heart-outline" size={16} color={colors.primary} />
          <Text style={styles.favoriteButtonText}>Marcar {activeTrip.driver.fullName.split(' ')[0]} como motorista preferido</Text>
        </TouchableOpacity>
      ) : null}
      {favoriteSaved ? (
        <View style={styles.favoriteConfirmed}>
          <Ionicons name="heart" size={14} color={colors.primary} />
          <Text style={styles.favoriteConfirmedText}>Motorista preferido guardado - vamos tenta-lo atribuir em proximas viagens.</Text>
        </View>
      ) : null}

      <TextInput
        style={styles.input}
        placeholder="Deixe um comentario (opcional)"
        placeholderTextColor={colors.textMuted}
        value={comment}
        onChangeText={setComment}
        multiline
      />

      <Button label="Concluir" onPress={handleFinish} style={{ marginTop: spacing.lg }} />

      <RewardsModal visible={rewardsVisible} onClose={() => setRewardsVisible(false)} />
    </ScreenContainer>
  );
}

import AsyncStorage from '@react-native-async-storage/async-storage';

const STORAGE_KEY = '@angotour/wallet_topups';

export const INITIAL_BALANCE_KZ = 0;

export interface WalletTopUp {
  id: string;
  dateIso: string;
  amountKz: number;
  reference: string;
  status: 'processing' | 'confirmed' | 'failed';
}

function generateReference(): string {
  return Math.floor(900000000 + Math.random() * 99999999).toString();
}

async function readTopUps(): Promise<WalletTopUp[]> {
  try {
    const raw = await AsyncStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

async function writeTopUps(topUps: WalletTopUp[]): Promise<void> {
  try {
    await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(topUps));
  } catch {
    // falha ao gravar nao deve bloquear o fluxo
  }
}

export async function getTopUps(): Promise<WalletTopUp[]> {
  const all = await readTopUps();
  return all.sort((a, b) => b.dateIso.localeCompare(a.dateIso));
}

/**
 * Carregar Carteira (seccao 14 do documento): gera uma referencia simulada
 * de Multicaixa Express. O estado comeca "processing" e e confirmado pouco
 * depois - nunca assumimos confirmacao instantanea (seccao 29).
 */
export async function startTopUp(amountKz: number): Promise<WalletTopUp> {
  const topUp: WalletTopUp = {
    id: `topup-${Date.now()}`,
    dateIso: new Date().toISOString(),
    amountKz,
    reference: generateReference(),
    status: 'processing',
  };
  const all = await readTopUps();
  await writeTopUps([topUp, ...all]);
  return topUp;
}

export async function confirmTopUp(id: string): Promise<void> {
  const all = await readTopUps();
  const updated = all.map((t) => (t.id === id ? { ...t, status: 'confirmed' as const } : t));
  await writeTopUps(updated);
}

export async function getConfirmedBalanceAdjustment(): Promise<number> {
  const all = await readTopUps();
  return all.filter((t) => t.status === 'confirmed').reduce((sum, t) => sum + t.amountKz, 0);
}

import { ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { WalletTopUp } from './entities/wallet-topup.entity';

@Injectable()
export class WalletService {
  constructor(
    @InjectRepository(WalletTopUp)
    private readonly topUps: Repository<WalletTopUp>,
  ) {}

  /** saldo = carregamentos confirmados - viagens concluidas pagas com a carteira */
  async summary(userId: string) {
    const list = await this.topUps.find({ where: { userId }, order: { createdAt: 'DESC' }, take: 50 });
    const confirmed = list.filter((t) => t.status === 'confirmed').reduce((s, t) => s + t.amountKz, 0);
    const spentRows = await this.topUps.query(
      `SELECT COALESCE(SUM("priceKz"),0) AS spent FROM trips WHERE "passengerId" = $1 AND status = 'completed' AND "paymentMethodType" = 'wallet'`,
      [userId],
    );
    const spent = Number(spentRows[0]?.spent ?? 0);
    return {
      balanceKz: confirmed - spent,
      topups: list.map((t) => ({ id: t.id, amountKz: t.amountKz, reference: t.reference, status: t.status, createdAt: t.createdAt })),
    };
  }

  async startTopUp(userId: string, amountKz: number): Promise<WalletTopUp> {
    const reference = `9${Math.floor(10000000 + Math.random() * 89999999)}`;
    return this.topUps.save(this.topUps.create({ userId, amountKz, reference, status: 'processing' }));
  }

  /**
   * SO DESENVOLVIMENTO: simula a confirmacao do Multicaixa. Em producao a confirmacao
   * vem de um webhook assinado do fornecedor de pagamentos - nunca do proprio cliente.
   * Bloqueado por omissao: exige ALLOW_DEV_TOPUP_CONFIRM=true explicito no .env, para
   * nunca depender de NODE_ENV estar (ou nao) configurado corretamente em producao.
   */
  async devConfirm(userId: string, id: string): Promise<WalletTopUp> {
    if (process.env.ALLOW_DEV_TOPUP_CONFIRM !== 'true') throw new ForbiddenException();
    const topUp = await this.topUps.findOne({ where: { id, userId } });
    if (!topUp) throw new NotFoundException('Carregamento nao encontrado.');
    topUp.status = 'confirmed';
    return this.topUps.save(topUp);
  }
}

import { BadRequestException, ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { randomUUID } from 'crypto';
import { promises as fs } from 'fs';
import * as path from 'path';
import { REQUIRED_IDENTITY_DOCUMENTS, UserDocument } from './entities/user-document.entity';

const UPLOAD_ROOT = path.resolve(process.cwd(), 'uploads', 'identity-docs');
const MAX_BYTES = 8 * 1024 * 1024;

// O tipo do ficheiro e decidido pelos primeiros bytes, nunca pelo que o cliente diz.
function detectImage(buf: Buffer): { ext: string; mime: string } | null {
  if (buf.length > 3 && buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return { ext: 'jpg', mime: 'image/jpeg' };
  if (buf.length > 8 && buf[0] === 0x89 && buf[1] === 0x50 && buf[2] === 0x4e && buf[3] === 0x47) return { ext: 'png', mime: 'image/png' };
  return null;
}

@Injectable()
export class IdentityDocumentsService {
  constructor(
    @InjectRepository(UserDocument)
    private readonly repo: Repository<UserDocument>,
  ) {}

  // ---------- lado do passageiro ----------

  async list(userId: string) {
    const docs = await this.repo.find({ where: { userId } });
    return REQUIRED_IDENTITY_DOCUMENTS.map((type) => {
      const d = docs.find((x) => x.type === type);
      return {
        type,
        status: d ? d.status : 'not_submitted',
        rejectionReason: d?.rejectionReason ?? null,
        updatedAt: d?.updatedAt ?? null,
      };
    });
  }

  async submit(userId: string, type: string, imageBase64: string) {
    if (!(REQUIRED_IDENTITY_DOCUMENTS as readonly string[]).includes(type)) {
      throw new BadRequestException('Tipo de documento invalido.');
    }
    const clean = imageBase64.replace(/^data:[^;]+;base64,/, '');
    const buf = Buffer.from(clean, 'base64');
    if (buf.length === 0 || buf.length > MAX_BYTES) {
      throw new BadRequestException('Imagem invalida ou maior que 8 MB.');
    }
    const kind = detectImage(buf);
    if (!kind) throw new BadRequestException('Formato invalido. Envia uma imagem JPEG ou PNG.');

    const existing = await this.repo.findOne({ where: { userId, type } });
    if (existing?.status === 'approved') {
      throw new BadRequestException('Este documento ja foi aprovado.');
    }

    const dir = path.join(UPLOAD_ROOT, userId);
    await fs.mkdir(dir, { recursive: true });
    const filePath = path.join(dir, `${type}-${randomUUID()}.${kind.ext}`);
    await fs.writeFile(filePath, buf);
    if (existing?.filePath) await fs.unlink(existing.filePath).catch(() => undefined);

    const doc = existing ?? this.repo.create({ userId, type });
    doc.status = 'pending';
    doc.filePath = filePath;
    doc.mimeType = kind.mime;
    doc.rejectionReason = null;
    doc.reviewedAt = null;
    await this.repo.save(doc);
    return { type, status: doc.status, rejectionReason: null, updatedAt: doc.updatedAt };
  }

  async areAllApproved(userId: string): Promise<boolean> {
    const docs = await this.repo.find({ where: { userId, status: 'approved' } });
    const approved = new Set(docs.map((d) => d.type));
    return REQUIRED_IDENTITY_DOCUMENTS.every((t) => approved.has(t));
  }

  // ---------- revisao (admin, mesmo guard do painel) ----------

  listForReview(status?: string) {
    const s = status === 'all' ? null : status && ['pending', 'approved', 'rejected'].includes(status) ? status : 'pending';
    return this.repo.query(
      `SELECT d.id, d."userId", d.type, d.status, d."rejectionReason", d."createdAt", d."updatedAt", u."fullName", u.phone
       FROM user_documents d JOIN users u ON u.id = d."userId"
       WHERE ($1::text IS NULL OR d.status = $1)
       ORDER BY d."updatedAt" ASC LIMIT 300`,
      [s],
    );
  }

  async info(id: string): Promise<{ userId: string; type: string }> {
    const doc = await this.repo.findOne({ where: { id } });
    if (!doc) throw new NotFoundException('Documento nao encontrado.');
    return { userId: doc.userId, type: doc.type };
  }

  async getFile(id: string): Promise<{ filePath: string; mimeType: string }> {
    const doc = await this.repo.findOne({ where: { id } });
    if (!doc) throw new NotFoundException('Documento nao encontrado.');
    const resolved = path.resolve(doc.filePath);
    if (!resolved.startsWith(UPLOAD_ROOT + path.sep)) throw new ForbiddenException();
    return { filePath: resolved, mimeType: doc.mimeType };
  }

  async approve(id: string) {
    const doc = await this.repo.findOne({ where: { id } });
    if (!doc) throw new NotFoundException('Documento nao encontrado.');
    doc.status = 'approved';
    doc.rejectionReason = null;
    doc.reviewedAt = new Date();
    await this.repo.save(doc);

    // Com os 3 aprovados, o passageiro passa a "verificado".
    const allApproved = await this.areAllApproved(doc.userId);
    if (allApproved) {
      await this.repo.query(
        `UPDATE user_profiles SET "isVerified" = true, "verificationStatus" = 'verified', "verifiedAt" = now() WHERE "userId" = $1`,
        [doc.userId],
      );
    }
    return { id: doc.id, type: doc.type, status: doc.status, allApproved };
  }

  async reject(id: string, reason: string) {
    const doc = await this.repo.findOne({ where: { id } });
    if (!doc) throw new NotFoundException('Documento nao encontrado.');
    doc.status = 'rejected';
    doc.rejectionReason = reason;
    doc.reviewedAt = new Date();
    await this.repo.save(doc);
    await this.repo.query(
      `UPDATE user_profiles SET "isVerified" = false, "verificationStatus" = 'rejected' WHERE "userId" = $1`,
      [doc.userId],
    );
    return { id: doc.id, type: doc.type, status: doc.status };
  }
}

@Injectable()
export class ProfileService {
  constructor(
    @InjectRepository(UserProfile)
    private readonly profiles: Repository<UserProfile>,
  ) {}

  async getOrCreate(userId: string): Promise<UserProfile> {
    let profile = await this.profiles.findOne({ where: { userId } });
    if (!profile) {
      await this.profiles.query(
        `INSERT INTO user_profiles ("userId","isVerified","angoPoints") VALUES ($1,false,0) ON CONFLICT ("userId") DO NOTHING`,
        [userId],
      );
      profile = await this.profiles.findOneOrFail({ where: { userId } });
    }
    return profile;
  }

  /**
   * Submete BI + selfie para revisao. NUNCA auto-aprova: fica "pending" ate um
   * revisor confirmar no Admin (Document Center). isVerified so passa a true
   * quando esse fluxo (ainda por construir) chamar approveVerification().
   */
  async verifyIdentity(userId: string): Promise<UserProfile> {
    // So se submete para revisao com os 3 documentos enviados (BI frente e verso, selfie).
    const docs: Array<{ ok: number }> = await this.profiles.query(
      'SELECT COUNT(*) FILTER (WHERE status <> \'rejected\')::int AS ok FROM user_documents WHERE "userId" = $1',
      [userId],
    );
    if (Number(docs[0]?.ok ?? 0) < 3) {
      throw new BadRequestException('Envia o BI (frente e verso) e a selfie antes de submeter a verificacao.');
    }
    const profile = await this.getOrCreate(userId);
    if (profile.verificationStatus !== 'verified') {
      profile.verificationStatus = 'pending';
    }
    return this.profiles.save(profile);
  }

  async approveVerification(userId: string): Promise<UserProfile> {
    const profile = await this.getOrCreate(userId);
    profile.isVerified = true;
    profile.verificationStatus = 'verified';
    profile.verifiedAt = new Date();
    return this.profiles.save(profile);
  }
}
